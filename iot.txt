code:
#include <WiFi.h>
#include <DHT.h>
#include <ThingSpeak.h>

const char* ssid = "YOUR_NETWORK_NAME";
const char* password = "NETWORK_PASSWORD";
unsigned long myChannelNumber = YOUR_CHANNEL_ID;
const char* myWriteAPIKey = "WRITE_API_KEY";

#define DHTPIN 23
#define DHTTYPE DHT11

DHT dht(DHTPIN, DHTTYPE);
WiFiClient client;

void setup() {
  Serial.begin(115200);
  delay(3000);
  dht.begin();
  ThingSpeak.begin(client);

  Serial.println("\n--- Initializing Library Integration Protocol ---");
  WiFi.disconnect(true, true);
  delay(1000);
  WiFi.mode(WIFI_STA);
  WiFi.begin(ssid, password);

  Serial.print("Connecting to network: ");
  Serial.println(ssid);

  while (WiFi.status() != WL_CONNECTED) {
    delay(1000);
    Serial.print(".");
  }

  Serial.println("\n Successfully Authenticated with Network!");
  Serial.print("Assigned IP Address: ");
  Serial.println(WiFi.localIP());
}

void loop() {
  if (WiFi.status() == WL_CONNECTED) {
    float temperature = dht.readTemperature();

    if (isnan(temperature)) {
      Serial.println(" Hardware Error: Check D23 data wire connection.");
      delay(3000);
      return;
    }

    Serial.print("\n Current Temperature: ");
    Serial.print(temperature);
    Serial.println(" °C");
    Serial.println("Routing data payload to ThingSpeak cloud via core library...");

    int responseCode = ThingSpeak.writeField(myChannelNumber, 1, temperature, myWriteAPIKey);

    if (responseCode == 200)
      Serial.println(" Cloud Update Verified! Dashboard graph updating successfully.");
    else {
      Serial.print(" Connection Blocked. ThingSpeak Error Code: ");
      Serial.println(responseCode);
      Serial.println("Note: If code is -301, Windows firewall is filtering outbound device packets.");
    }
  } else {
    Serial.println(" Connection dropped. Re-authenticating...");
    WiFi.begin(ssid, password);
  }

  Serial.println("Waiting 20 seconds for the next data push cycle...");
  delay(20000);
}

connection vcc-5v,gnd-gnd,trig-6,echo-5
buzzer + to 4 ,gnd to gnd,led to resistor to 9,led gnd to gnd

code:

int TRIG = 6;
int ECHO = 5;
int BUZZER = 4;
int RED_LED = 9;
int GREEN_LED = 10;
int WARNING_DISTANCE = 50;

void setup() {
  Serial.begin(9600);

  pinMode(TRIG, OUTPUT);
  pinMode(ECHO, INPUT);
  pinMode(BUZZER, OUTPUT);
  pinMode(RED_LED, OUTPUT);
  pinMode(GREEN_LED, OUTPUT);

  Serial.println("System Ready...");
}

void loop() {
  digitalWrite(TRIG, LOW);
  delayMicroseconds(2);
  digitalWrite(TRIG, HIGH);
  delayMicroseconds(10);
  digitalWrite(TRIG, LOW);

  long duration = pulseIn(ECHO, HIGH);
  int distance = duration * 0.034 / 2;

  Serial.print("Distance: ");
  Serial.print(distance);
  Serial.println(" cm");

  if (distance > 0 && distance <= WARNING_DISTANCE) {
    digitalWrite(RED_LED, HIGH);
    digitalWrite(GREEN_LED, LOW);
    tone(BUZZER, 1000);
    Serial.println("INTRUSION DETECTED");
  } else {
    digitalWrite(RED_LED, LOW);
    digitalWrite(GREEN_LED, HIGH);
    noTone(BUZZER);
    Serial.println("AREA SECURE");
  }

  delay(100);
}

exp 10: motor
connection ENA to 5v,in1-5,in2-6,out1-dc motor ,out2 dc motor,gnd-gnd
blue led + to resistor to 3,- to gnd
red led + to resistor to 4,- to gnd
switch terminal 1 to pin 2, terminal 2 to 5v

code:

int inputPin = 1;
int bluePin = 3;
int redPin = 4;
int motorPin1 = 5;
int motorPin2 = 6;

int dir = LOW;
int prevState = LOW;

void setup() {
  pinMode(inputPin, INPUT);
  pinMode(bluePin, OUTPUT);
  pinMode(redPin, OUTPUT);
  pinMode(motorPin1, OUTPUT);
  pinMode(motorPin2, OUTPUT);
}

void loop() {
  int currentState = digitalRead(inputPin);

  if (currentState == HIGH && prevState == LOW)
    dir = !dir;

  prevState = currentState;

  if (dir == HIGH) {
    digitalWrite(motorPin1, HIGH);
    digitalWrite(motorPin2, LOW);
    digitalWrite(bluePin, LOW);
    digitalWrite(redPin, HIGH);
  } else {
    digitalWrite(motorPin1, LOW);
    digitalWrite(motorPin2, HIGH);
    digitalWrite(bluePin, HIGH);
    digitalWrite(redPin, LOW);
  }
}

int trigPin = 5, echoPin = 18;
WiFiClient client;

void setup() {
  Serial.begin(115200);
  pinMode(trigPin, OUTPUT);
  pinMode(echoPin, INPUT);

  ThingSpeak.begin(client);
  WiFi.begin(ssid, password);

  while (WiFi.status() != WL_CONNECTED) delay(1000);

void loop() {
  if (WiFi.status() == WL_CONNECTED) {
    digitalWrite(trigPin, LOW);
    delayMicroseconds(2);
    digitalWrite(trigPin, HIGH);
    delayMicroseconds(10);
    digitalWrite(trigPin, LOW);

    long duration = pulseIn(echoPin, HIGH);

    if (duration == 0) return;

    float distance = duration * 0.0343 / 2;

    Serial.print("Distance: ");
    Serial.print(distance);
    Serial.println(" cm");

    ThingSpeak.writeField(channelID, 1, distance, apiKey);
  } else {
    WiFi.begin(ssid, password);
  }

  delay(20000);
}
